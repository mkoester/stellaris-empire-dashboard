# All lines go to game.log as SDASH|2|<date>|…, parsed by src/lib/server/feed.ts — change both
# together, and bump the format number. Human players only.
# Resource keys from 4.5.2 common/strategic_resources/00_strategic_resources.txt.
# The lines are spelled out per resource: a $RES$ parameter inside the quoted log string came out
# one character off in 4.5.2 (`|$RES$|` → `energy$`; `$X|y$` is the parameter-default syntax).

# Monthly: a new timeline id, then stockpile and income of every resource, then an end marker so
# the reader knows the month is complete.
country_event = {
	id = sdash.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
	}

	immediate = {
		sdash_roll_timeline = yes
		# 0.4.0: every Galactic Community member's diplomatic weight, which the game does not save —
		# into the country variable sdash_dw, so every save carries it (the dashboard's vote sums);
		# no log line. Trigger and effect from 4.5.2 script_documentation (triggers.log, effects.log).
		every_country = {
			limit = { is_galactic_community_member = yes }
			export_trigger_value_to_variable = { trigger = diplomacy_weight variable = sdash_dw }
		}
		# 0.5.0: storage — each resource's maximum, which the game does not save: ours with every
		# resource (sdash_m_<res>, in sdash_export_resource), the tradable ones of every empire we
		# talk to (their room in a trade). No log line. Effect from 4.5.2 effects.log.
		every_country = {
			limit = { has_communications = root }
			sdash_export_tradable_maximums = yes
		}

# Starts the tick unless a chain is already running.
country_event = {
	id = sdash.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		NOT = { has_country_flag = sdash_tick_alive }
	}

	immediate = {
		set_timed_country_flag = { flag = sdash_tick_alive days = 3 }
		country_event = { id = sdash.2 days = 1 }
	}
}

# A save was loaded: its timeline id names the month it was saved in, i.e. the branch point.
event = {
	id = sdash.4
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_playable_country = {
			limit = { is_ai = no }
			if = {
				limit = { is_variable_set = sdash_tl }
				log = "SDASH|2|[GetDate]|load|[This.sdash_tl]"
			}
			else = {
				log = "SDASH|2|[GetDate]|load|0"
			}
			country_event = { id = sdash.3 }
		}
	}
}
